import { pathToFileURL } from 'node:url';
export function formatPhone(input) {
  const d = String(input ?? '').replace(/\D/g, '');
  if (d.startsWith('02')) {
    if (d.length === 9) return `02-${d.slice(2, 5)}-${d.slice(5)}`;
    if (d.length === 10) return `02-${d.slice(2, 6)}-${d.slice(6)}`;

  }
  if (d.startsWith('0')) {
    if (d.length === 10) return `${d.slice(0, 3)}-${d.slice(3, 6)}-${d.slice(6)}`;
    if (d.length === 11) return `${d.slice(0, 3)}-${d.slice(3, 7)}-${d.slice(7)}`;
  }
  return null;
}

--- reference
import { pathToFileURL } from 'node:url';
export function parseCsvLine(line) {
  const out = []; let cur = ''; let q = false;
  for (let i = 0; i < line.length; i++) {
    const ch = line[i];
    if (q) {
      if (ch === '"') { if (line[i + 1] === '"') { cur += '"'; i++; } else q = false; }
      else cur += ch;
    } else if (ch === '"') q = true;
    else if (ch === ',') { out.push(cur); cur = ''; }
    else cur += ch;
  }
  out.push(cur);
  return out;
}

--- reference
import { pathToFileURL } from 'node:url';
export function paginate(total, page, size) {
  const pages = Math.max(1, Math.ceil(total / size));
  const p = Math.min(pages, Math.max(1, Math.floor(page) || 1));
  if (total === 0) return { pages, page: p, from: 0, to: 0 };
  return { pages, page: p, from: (p - 1) * size + 1, to: Math.min(total, p * size) };
}

export function addBusinessDays(dateStr, n) {
  const [y, m, d] = dateStr.split('-').map(Number);

--- reference
import { pathToFileURL } from 'node:url';
const toMin = (t) => { const [h, m] = t.split(':').map(Number); return h * 60 + m; };
const fmt = (x) => `${String(Math.floor(x / 60)).padStart(2, '0')}:${String(x % 60).padStart(2, '0')}`;
export function slots(start, end, minutes) {
  const out = []; const e = toMin(end);
  for (let t = toMin(start); t + minutes <= e; t += minutes) out.push(fmt(t));
  return out;
}
